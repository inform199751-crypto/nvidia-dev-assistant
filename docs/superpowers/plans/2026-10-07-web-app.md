> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把現有的指令列 agent 包成本機執行的 ChatGPT / Claude 式網頁 App:選資料夾或上傳、用中文問、即時看到每一步、能追問、長任務在背景跑。

**Architecture:** FastAPI 後端 + 單頁網頁(原生 JS,無建置工具)。每一則訊息是一「輪」,在執行緒池跑既有的 `run_agent`,透過事件回呼把每一步存進記憶體並同步寫 JSON 檔;網頁用 SSE 先補送已有事件再推新事件。既有程式只改兩處:工具改成每個任務綁定自己的資料夾、`run_agent` 增加事件 / 追問 / 停止三個參數。

**Tech Stack:** Python 3.13、FastAPI、uvicorn、python-multipart、LangChain core 1.6、langchain-nvidia-ai-endpoints 1.4.3、pytest + FastAPI TestClient;前端 marked 12.0.2 + DOMPurify 3.1.6(放進 repo)。

class ScriptedChat(BaseChatModel):
    """照劇本回答的假模型;記錄每次收到的訊息與自己的角色。"""
    role: str = ""
    script: Any = None  # 用 Any,pydantic 才不會複製 list,多個角色共用同一份劇本
    seen: Any = None

    def _generate(self, messages, stop=None, run_manager=None, **kwargs: Any) -> ChatResult:
        self.seen.append((self.role, list(messages)))
        return ChatResult(generations=[ChatGeneration(message=self.script.pop(0))])

    def bind_tools(self, tools, **kwargs):
        return self

    @property
    def _llm_type(self) -> str:
        return "scripted"


def scripted_registry(script: list[AIMessage], config: dict = CONFIG):
    shared = {"script": script, "seen": []}

    def factory(spec, **_):
        return ScriptedChat(role=spec.role, script=shared["script"], seen=shared["seen"])

    return ModelRegistry(copy.deepcopy(config), backend="nim", chat_factory=factory), shared["seen"]


def ai(content="", tool_calls=(), tokens=100, reasoning=None):
    kw = {"reasoning_content": reasoning, "_reasoning_api_fields": ["reasoning_content"]} if reasoning else {}
    return AIMessage(content=content, tool_calls=list(tool_calls), additional_kwargs=kw,
                     usage_metadata={"input_tokens": tokens, "output_tokens": 0, "total_tokens": tokens})


class InlineExecutor:
    """測試用:submit 時直接在呼叫端執行,呼叫回來時這一輪已經跑完。"""

    def submit(self, fn, *args, **kwargs):
        fn(*args, **kwargs)


class ManualExecutor:
    """測試用:先收著,測試自己決定什麼時候跑(用來測排隊、取消、忙碌)。"""

    def __init__(self):
        self.jobs = []

    def submit(self, fn, *args, **kwargs):
        self.jobs.append((fn, args, kwargs))

    def run_all(self):
        while self.jobs:
            fn, args, kwargs = self.jobs.pop(0)
            fn(*args, **kwargs)

"""工具綁定資料夾:網頁同時跑多個任務、各自在不同資料夾時不能互相覆蓋。"""
from __future__ import annotations

from concurrent.futures import ThreadPoolExecutor

import pytest

import tools


def by_name(toolset):
    return {t.name: t for t in toolset}


def test_each_toolset_reads_its_own_folder(tmp_path):
    a, b = tmp_path / "a", tmp_path / "b"
    a.mkdir()
    b.mkdir()
    (a / "x.py").write_text("print('A')\n", encoding="utf-8")
    (b / "x.py").write_text("print('B')\n", encoding="utf-8")
    ta, tb = by_name(tools.make_tools(a)), by_name(tools.make_tools(b))
    assert "print('A')" in ta["read_file"].invoke({"path": "x.py"})
    assert "print('B')" in tb["read_file"].invoke({"path": "x.py"})
    assert tb["search_code"].invoke({"pattern": "print"}).startswith("x.py:1: print('B')")


def test_toolsets_stay_isolated_across_threads(tmp_path):
    roots = []
    for i in range(4):
        d = tmp_path / f"w{i}"
        d.mkdir()
        (d / "id.txt").write_text(f"ws{i}\n", encoding="utf-8")
        roots.append(d)
    sets = [by_name(tools.make_tools(d)) for d in roots]
    order = [0, 1, 2, 3] * 5
    with ThreadPoolExecutor(4) as ex:
        outs = list(ex.map(lambda i: sets[i]["read_file"].invoke({"path": "id.txt"}), order))
    assert all(f"ws{i}" in out for i, out in zip(order, outs))


def test_fixed_toolset_ignores_global_workspace(tmp_path, monkeypatch):
    fixed, other = tmp_path / "fixed", tmp_path / "other"
    fixed.mkdir()
    other.mkdir()
    (fixed / "f.py").write_text("FIXED\n", encoding="utf-8")
    t = by_name(tools.make_tools(fixed))
    monkeypatch.setattr(tools, "WORKSPACE", other)
    assert "FIXED" in t["read_file"].invoke({"path": "f.py"})


def test_fixed_toolset_blocks_secrets_and_escape(tmp_path):

```python
"""Agent 工具。原則:小而精準的輸出 + 錯誤回給模型自己修正 + 不直接改檔。

工具由 make_tools(資料夾) 產生,每組綁定自己的資料夾 —— 網頁同時跑多個任務、各自在不同資料夾時,
才不會互相覆蓋。模組層的 read_file / ALL_TOOLS 是「跟著全域 WORKSPACE 走」的預設組,給指令列用。
"""
from __future__ import annotations

import os
import re
import shutil
import subprocess
import tempfile
from datetime import datetime
from pathlib import Path

from langchain_core.tools import BaseTool, tool

WORKSPACE = Path(os.getenv("AGENT_WORKSPACE", ".")).resolve()
SANDBOX_IMAGE = os.getenv("SANDBOX_IMAGE", "dev-assistant-sandbox")  # sandbox/Dockerfile 建的(含 pytest)
MAX_READ_LINES = 400


def configure(workspace: str | Path | None = None, sandbox_image: str | None = None) -> None:
    """讀完 .env 之後要再呼叫一次 —— 模組載入時 .env 還沒讀,上面兩個值會是預設值。"""
    global WORKSPACE, SANDBOX_IMAGE
    if workspace is not None:
        WORKSPACE = Path(workspace).resolve()
    if sandbox_image:
        SANDBOX_IMAGE = sandbox_image


_SECRET_SUFFIXES = {".pem", ".key", ".p12", ".pfx", ".jks", ".keystore"}
_SECRET_PREFIXES = ("id_rsa", "id_ed25519", "id_ecdsa")


def _is_secret(p: Path) -> bool:
    """金鑰、憑證、.env 一律不給模型看 —— 讀到就等於把金鑰送出公司。"""
    name = p.name.lower()
    if name == ".env" or (name.startswith(".env.") and not name.endswith((".example", ".sample", ".template"))):
        return True
    return p.suffix.lower() in _SECRET_SUFFIXES or name.startswith(_SECRET_PREFIXES)


def safe_path(root: Path, rel: str) -> Path:
    """把相對路徑解析到 root 底下;越界或碰到機密檔就丟錯(錯誤會回給模型)。"""
    root = Path(root).resolve()
    p = (root / rel).resolve()
    if not p.is_relative_to(root):
        raise ValueError(f"路徑 {rel} 超出工作區,拒絕存取")
    if _is_secret(p) or ".git" in p.relative_to(root).parts:
        raise PermissionError(f"{rel} 是機密檔(金鑰 / 憑證 / .env / .git),agent 不能讀")
    return p


def _safe_path(rel: str) -> Path:
    return safe_path(WORKSPACE, rel)


# 暫存複本不帶 .env 和金鑰檔:沙箱裡的測試輸出會回到模型手上
_COPY_IGNORE = shutil.ignore_patterns(".agent_patches", ".git", ".venv", "__pycache__", "node_modules",
                                      ".env", ".env.local", ".env.prod*", "*.pem", "*.key", "id_rsa*")


def _run_sandbox(root: Path, target: str, image: str | None = None) -> str:
    """在隔離的 Docker 容器裡跑 pytest:無網路、唯讀掛載、限 CPU / 記憶體 / 行程數 / 時間。"""
    if not shutil.which("docker"):
        return "錯誤:這台機器沒有 Docker,沙箱無法啟動。為了安全,不會改在主機上直接執行程式碼。"
    cmd = [
        "docker", "run", "--rm", "--network", "none", "--read-only", "--tmpfs", "/tmp",
        "--memory", "1g", "--cpus", "1", "--pids-limit", "256",
        "-e", "PYTHONDONTWRITEBYTECODE=1", "-v", f"{root}:/ws:ro", "-w", "/ws",
        image or SANDBOX_IMAGE, "python", "-m", "pytest", "-q", "-p", "no:cacheprovider", target,
    ]
    try:
        r = subprocess.run(cmd, capture_output=True, text=True, timeout=180)
    except subprocess.TimeoutExpired:
        return "錯誤:測試超過 180 秒被中止(可能有無窮迴圈或死結)"
    return f"exit code {r.returncode}\n{r.stdout}\n{r.stderr}".strip()


def make_tools(workspace: str | Path | None = None, sandbox_image: str | None = None) -> list[BaseTool]:
    """產生一組綁定資料夾的工具。workspace=None 時每次呼叫才讀全域 WORKSPACE(指令列與舊測試用)。"""
    fixed = Path(workspace).resolve() if workspace is not None else None

    def root() -> Path:
        return fixed if fixed is not None else WORKSPACE

    def check_target(target: str) -> str | None:
        if target.startswith("-"):
            return "錯誤:target 必須是路徑,不能是參數"
        safe_path(root(), target)
        return None

    @tool
    def read_file(path: str, start_line: int = 1, end_line: int | None = None) -> str:
        """讀取工作區檔案的指定行號範圍,每行前面附行號。一次最多 400 行;大檔請先看大綱或 search_code 再分段讀。"""
        lines = safe_path(root(), path).read_text(encoding="utf-8", errors="replace").splitlines()
        start = max(start_line, 1)
        end = min(end_line or len(lines), len(lines), start + MAX_READ_LINES - 1)
        body = "\n".join(f"{n:>5}| {lines[n - 1]}" for n in range(start, end + 1))
        more = f"\n[檔案共 {len(lines)} 行,已顯示 {start}-{end}]" if end < len(lines) or start > 1 else ""
        return body + more

    @tool
    def search_code(pattern: str, glob: str = "**/*.py", max_hits: int = 50) -> str:
        """在工作區用正規表示式搜尋,回傳「檔案:行號: 內容」。先搜再讀,不要整檔讀。"""
        base = root()
        rx = re.compile(pattern)
        hits: list[str] = []
        for f in base.glob(glob):
            if (not f.is_file() or _is_secret(f)
                    or any(part.startswith(".") for part in f.relative_to(base).parts)):
                continue
            for n, line in enumerate(f.read_text(encoding="utf-8", errors="replace").splitlines(), 1):
                if rx.search(line):
                    hits.append(f"{f.relative_to(base).as_posix()}:{n}: {line.strip()[:200]}")
                    if len(hits) >= max_hits:
                        return "\n".join(hits) + f"\n[已達 {max_hits} 筆上限,請把 pattern 寫得更精確]"
        return "\n".join(hits) or "沒有符合的結果"

    @tool
    def run_tests(target: str = "tests") -> str:
        """在 Docker 沙箱跑 pytest,測的是「目前的原始碼」(提出的 patch 不會套用)。改 bug 前先用它重現問題。"""
        return check_target(target) or _run_sandbox(root(), target, image=sandbox_image)

    @tool
    def propose_patch(path: str, unified_diff: str, reason: str, test_target: str = "tests") -> str:
        """提出修改:輸入 unified diff,存成 .patch 等人工審核,原始檔案不會被改動。
        會先檢查 diff 能不能套用,再把 patch 套到工作區的暫存複本上、在沙箱跑 test_target,回傳套用後的真實測試結果。"""
        base = root()
        safe_path(base, path)
        if err := check_target(test_target):
            return err
        if not shutil.which("git"):
            return "錯誤:這台機器沒有 git,無法驗證 patch"
        out_dir = base / ".agent_patches"
        out_dir.mkdir(exist_ok=True)
        patch_file = out_dir / f"{datetime.now():%Y%m%d_%H%M%S}_{Path(path).name}.patch"
        patch_file.write_text(f"# {reason}\n{unified_diff.rstrip()}\n", encoding="utf-8", newline="\n")
        r = subprocess.run(["git", "apply", "--check", str(patch_file)],
                           cwd=base, capture_output=True, text=True)
        if r.returncode != 0:
            patch_file.unlink()
            return f"diff 套不上去,請依錯誤修正後重送:\n{r.stderr.strip()}"

        # 套到暫存複本上驗證;正式環境大 repo 改用 git worktree,不要整包複製
        with tempfile.TemporaryDirectory(prefix="agent_verify_") as tmp:
            copy = Path(tmp) / "ws"
            shutil.copytree(base, copy, ignore=_COPY_IGNORE)
            subprocess.run(["git", "apply", str(patch_file)], cwd=copy, check=True, capture_output=True)
            result = _run_sandbox(copy, test_target, image=sandbox_image)
        saved = patch_file.relative_to(base).as_posix()
        return f"已存成 {saved},等工程師審核後套用。\n套用此 patch 後的測試結果:\n{result}"

    return [read_file, search_code, run_tests, propose_patch]


ALL_TOOLS = make_tools()
read_file, search_code, run_tests, propose_patch = ALL_TOOLS
```

"""對內研發助理:核心 agent 迴圈。

自己寫迴圈而不用現成的 create_agent,是為了在「每一步送出前」插入 token 檢查、
歷史瘦身與模型升級 —— 這三件事就是防 OOM 和控成本的位置。
"""
from __future__ import annotations

import argparse
import logging
import os
import re
from dataclasses import dataclass, field
from pathlib import Path
from typing import Callable

from langchain_core.messages import BaseMessage, HumanMessage, SystemMessage, ToolMessage

from llm_registry import ModelRegistry
from token_guard import (TokenQuota, compact_history, estimate_tokens, message_tokens,
                         outline, strip_stale_reasoning, strip_think_text, truncate_middle)
import tools as toolbox

log = logging.getLogger("dev_assistant")

# 指令用英文寫(各家程式碼模型的指令微調資料以英文為主),回答語言另外指定。
SYSTEM_PROMPT = """You are an internal senior software engineer assistant. Reply in Traditional Chinese (zh-TW); keep code, identifiers and error messages in their original language.

## Workflow for bugs and code changes
1. Locate: use search_code / read_file to find the relevant lines. Do not guess what code you have not read.
2. Reproduce: run run_tests (or reason from the given traceback) before changing anything.
3. Hypothesize: state the root cause in one or two sentences, citing path:line.
4. Fix and verify: call propose_patch with a minimal unified diff (paths relative to workspace, a/ b/ prefixes). It applies the patch to a scratch copy, runs the tests and returns the real post-patch result. Do not refactor unrelated code.
5. If the post-patch tests still fail, read that output and send a revised patch. Note: run_tests always tests the ORIGINAL code, so do not use it to verify a patch.

## Rules
- Cite evidence as path:line. If something is uncertain or not visible in the files, say so instead of inventing APIs or behavior.
- Never state test results that did not come from a tool output in this conversation.
- Prefer small targeted reads over reading whole files. Large files come with an outline; read only the ranges you need.
- Tool errors are feedback: read them and correct your call; do not repeat the same failing call.
- For documentation tasks: structure as Overview → Architecture → Key flows → API/Config → Known limitations."""

WHOLE_NOTE = "<note>所有相關檔案都已完整附在上面。這次沒有工具可用,請直接根據附檔內容完成任務。</note>"
OMITTED_NOTE = "<note>前文附檔內容已省略;需要時用 read_file 讀取。</note>"
_FILE_BLOCK = re.compile(r'<file path="([^"]+)" lines="[^"]*">\n.*?\n</file>', re.S)
_FAILED_PREFIXES = ("錯誤", "工具執行錯誤", "diff 套不上去")


@dataclass
class AgentResult:
    answer: str
    role: str
    steps: int
    tokens_used: int
    messages: list[BaseMessage] = field(default_factory=list)  # 完整訊息串,追問時當前文
    stopped: bool = False


def _numbered(text: str) -> str:
    return "\n".join(f"{n:>5}| {line}" for n, line in enumerate(text.splitlines(), 1))


def build_user_prompt(task: str, attachments: list[str], inline_limit: int, whole: bool,
                      root: Path | None = None) -> str:
    """長內容放前面、問題放最後(長文本模型對結尾的指令最敏感,也讓前綴快取命中)。

    whole=False:小檔內嵌全文,大檔只給大綱 + 工具自己讀(省 token,幾乎所有改 bug 情境都夠用)。
    whole=True :全部內嵌,交給路由決定要不要升級到長文本模型(寫整份技術文件、全檔審查)。
    root:附檔所在資料夾;None 就用全域 WORKSPACE(指令列)。
    """
    base = root if root is not None else toolbox.WORKSPACE
    parts = []
    for rel in attachments:
        text = toolbox.safe_path(base, rel).read_text(encoding="utf-8", errors="replace")  # 附檔也要過機密檔檢查
        n_lines = text.count("\n") + 1
        if whole or estimate_tokens(text) <= inline_limit:
            parts.append(f'<file path="{rel}" lines="1-{n_lines}">\n{_numbered(text)}\n</file>')
        else:
            parts.append(f'<file_outline path="{rel}" total_lines="{n_lines}">\n{outline(text, rel)}\n'
                         f"</file_outline>\n(檔案太大未內嵌,請用 read_file 讀需要的行號範圍)")
    if whole:  # 整檔模式不給工具:檔案都在上面了,給工具只會讓模型分段重讀、每步重送全部內容
        parts.append(WHOLE_NOTE)
    parts.append(f"<task>\n{task}\n</task>")
    return "\n\n".join(parts)


def omit_attachments(messages: list[BaseMessage]) -> list[BaseMessage]:
    """追問時不重送前文的附檔全文(整份文件模式一次就是十幾萬 tokens),只留檔名;需要再用工具讀。"""
    out: list[BaseMessage] = []
    for m in messages:
        if isinstance(m, HumanMessage) and isinstance(m.content, str) and '<file path="' in m.content:
            content = _FILE_BLOCK.sub(r'<file path="\1" omitted="true"/>', m.content)
            content = content.replace(WHOLE_NOTE, OMITTED_NOTE)
            if OMITTED_NOTE not in content:
                content = content.replace("<task>", f"{OMITTED_NOTE}\n\n<task>", 1)
            m = m.model_copy(update={"content": content})
        out.append(m)
    return out


def _patch_event(args: dict, output: str) -> dict:
    """把 propose_patch 的參數(diff)和結果(套用後的測試輸出)組成畫面要的修正事件。"""
    saved = re.match(r"已存成 (\S+?),", output)
    test_output = output.split("套用此 patch 後的測試結果:\n", 1)[-1]
    return {"type": "patch", "path": args.get("path", ""), "diff": args.get("unified_diff", ""),
            "patch_file": saved.group(1) if saved else "", "test_output": test_output[:4000],
            "passed": test_output.startswith("exit code 0")}


def run_agent(task: str, *, registry: ModelRegistry, quota: TokenQuota, user: str = "cli",
              attachments: list[str] = (), role: str | None = None, thinking: bool | None = None,
              whole: bool = False, tools: list | None = None, workspace: Path | None = None,
              history: list[BaseMessage] | None = None,
              on_event: Callable[[dict], None] | None = None,
              should_stop: Callable[[], bool] | None = None) -> AgentResult:
    """跑一輪 agent。

    tools=None 時用 make_tools(workspace) 產生綁定資料夾的工具;workspace=None 就跟著全域 WORKSPACE(指令列)。
    history:前幾輪的完整訊息串(追問用),附檔全文會先被 omit_attachments 拿掉。
    on_event:每一步回報事件給網頁;should_stop:每步送出前檢查,回 True 就停。
    """
    emit = on_event or (lambda event: None)
    limits = registry.limits
    if tools is None:
        tools = toolbox.make_tools(workspace)
    if whole:
        tools = []
    prompt = HumanMessage(build_user_prompt(task, list(attachments), limits["inline_file_max_tokens"],
                                            whole, root=workspace))
    messages: list[BaseMessage] = (omit_attachments(list(history)) if history
                                   else [SystemMessage(SYSTEM_PROMPT)]) + [prompt]
    role = registry.pick_role(prompt_tokens=message_tokens(messages), needs_tools=bool(tools), requested=role)
    llm = registry.agent_model(role, tools, thinking=thinking)
    tool_map = {t.name: t for t in tools}
    used = 0
    scale = 1.0  # 估算偏低時,用伺服器回報的實際 input tokens 往上修正
    emit({"type": "start", "role": role, "model": registry.spec(role).model,
          "mode": "whole" if whole else "fix", "est_tokens": message_tokens(messages)})

    def sized() -> int:
        return int(message_tokens(messages) * scale)

    with quota.slot(user):
        for step in range(1, limits["max_agent_steps"] + 1):
            if should_stop and should_stop():
                return AgentResult("[已停止]", role, step - 1, used, messages, stopped=True)
            # 每一步送出前的三道檢查:清掉舊思考 → 瘦身 → 還放不下就升級模型
            messages = strip_stale_reasoning(messages)
            budget = registry.budget(role)
            if sized() > budget:
                messages = compact_history(messages, int(budget / scale))
            if (tokens := sized()) > budget:
                new_role = registry.pick_role(prompt_tokens=tokens, needs_tools=bool(tools))  # 放不下會丟 ContextTooLarge
                log.info("對話約 %s tokens 超過 %s 的預算,升級到 %s", tokens, role, new_role)
                emit({"type": "model_switch", "from_role": role, "to_role": new_role,
                      "reason": f"內容約 {tokens:,} tokens,超過 {role} 的上限"})
                role, llm = new_role, registry.agent_model(new_role, tools, thinking=thinking)
            quota.check(user, sized() + registry.spec(role).max_output)  # 每步都檢查,不是只在開始
            emit({"type": "step", "step": step, "role": role, "model": registry.spec(role).model})

            estimated = message_tokens(messages)
            ai = llm.invoke(messages)
            usage = ai.usage_metadata or {}
            if usage.get("input_tokens") and estimated and usage["input_tokens"] / estimated > scale * 1.05:
                scale = usage["input_tokens"] / estimated
                log.info("估算偏低:實際 %s / 估算 %s,之後的預算按 %.2f 倍計", usage["input_tokens"], estimated, scale)
            step_tokens = usage.get("total_tokens", 0)
            used += step_tokens
            quota.charge(user, step_tokens)
            messages.append(ai)
            calls = ", ".join(f"{c['name']}({str(c['args'])[:60]})" for c in ai.tool_calls) or "(回答)"
            log.info("step %d  role=%s  tokens=%d  %s", step, role, step_tokens, calls)

            if not ai.tool_calls:
                answer = strip_think_text(str(ai.content))
                emit({"type": "answer", "text": answer})
                return AgentResult(answer, role, step, used, messages)
            for call in ai.tool_calls:
                emit({"type": "tool_call", "step": step, "name": call["name"], "args": call["args"]})
                tool = tool_map.get(call["name"])
                try:
                    out = str(tool.invoke(call["args"])) if tool else f"錯誤:沒有 {call['name']} 這個工具"
                except Exception as e:  # 錯誤回給模型自己修正,不讓整個流程掛掉
                    out = f"工具執行錯誤:{type(e).__name__}: {e}"
                emit({"type": "tool_result", "step": step, "name": call["name"],
                      "ok": not out.startswith(_FAILED_PREFIXES), "preview": out[:4000]})
                if call["name"] == "propose_patch" and out.startswith("已存成"):
                    emit(_patch_event(call["args"], out))
                messages.append(ToolMessage(truncate_middle(out, limits["tool_output_max_tokens"]),
                                            tool_call_id=call["id"]))
    answer = f"[已達 {limits['max_agent_steps']} 步上限,停止。請把問題拆小再問]"
    emit({"type": "answer", "text": answer})
    return AgentResult(answer, role, limits["max_agent_steps"], used, messages)

"""網頁可選的資料夾:設定檔清單 + 上傳的暫存複本。

上傳一律先檢查再落地:路徑穿越、符號連結、超量整包拒收;機密檔與 .git 丟掉並告知。
"""
from __future__ import annotations

import io
import json
import shutil
import subprocess
import time
import uuid
import zipfile
from dataclasses import dataclass
from pathlib import Path, PurePosixPath

import yaml

from tools import _is_secret

MAX_TOTAL_BYTES = 20 * 1024 * 1024
MAX_FILES = 2000
MAX_FILE_BYTES = 2 * 1024 * 1024
TTL_SECONDS = 24 * 3600
TEXT_SUFFIXES = {
    ".py", ".pyi", ".js", ".jsx", ".ts", ".tsx", ".vue", ".java", ".kt", ".go", ".rs", ".c", ".h",
    ".cpp", ".hpp", ".cs", ".rb", ".php", ".swift", ".scala", ".sql", ".sh", ".ps1", ".md", ".rst",
    ".txt", ".yaml", ".yml", ".toml", ".ini", ".cfg", ".json", ".html", ".css", ".scss", ".j2",
    ".jinja", ".xml", ".gradle",
}
_SKIP_DIRS = {".git", "__MACOSX"}


class UploadRejected(ValueError):
    """整包拒收,訊息會直接顯示給使用者。"""


@dataclass(frozen=True)
class Workspace:
    id: str
    name: str
    path: Path
    kind: str  # "config" | "upload"
    created_at: float = 0.0

    def to_dict(self) -> dict:
        return {"id": self.id, "name": self.name, "kind": self.kind, "path": str(self.path)}


def _clean_rel(name: str) -> PurePosixPath | None:
    """回傳安全的相對路徑;機密檔、.git、macOS 雜檔回 None(丟掉);危險路徑整包拒收。"""
    name = name.replace("\\", "/")
    p = PurePosixPath(name)
    if name.startswith("/") or (len(name) > 1 and name[1] == ":") or ".." in p.parts:
        raise UploadRejected(f"路徑可疑,整包拒收:{name}")
    if not p.parts or any(part in _SKIP_DIRS for part in p.parts) or _is_secret(Path(p.name)):
        return None
    return p


class WorkspaceRegistry:
    def __init__(self, config_path: Path, base_dir: Path, uploads_dir: Path, ttl_seconds: int = TTL_SECONDS):
        self.base_dir = Path(base_dir).resolve()
        self.uploads_dir = Path(uploads_dir)
        self.ttl = ttl_seconds
        cfg = yaml.safe_load(Path(config_path).read_text(encoding="utf-8")) if Path(config_path).exists() else {}
        self._config: list[Workspace] = []
        for item in (cfg or {}).get("workspaces", []):
            path = Path(item["path"])
            path = (path if path.is_absolute() else self.base_dir / path).resolve()
            self._config.append(Workspace(id=item["name"], name=item["name"], path=path, kind="config"))

    # ---------- 查詢 ----------
    def list(self) -> list[Workspace]:
        out = [w for w in self._config if w.path.is_dir()]
        if self.uploads_dir.exists():
            for d in sorted(self.uploads_dir.iterdir()):
                w = self._load_upload(d)
                if w and time.time() - w.created_at < self.ttl:
                    out.append(w)
        return out

    def get(self, workspace_id: str) -> Workspace:
        for w in self.list():
            if w.id == workspace_id:
                return w
        raise KeyError(workspace_id)

    def _load_upload(self, d: Path) -> Workspace | None:
        meta = d / "meta.json"
        if not meta.exists():
            return None
        m = json.loads(meta.read_text(encoding="utf-8"))
        return Workspace(id=d.name, name=f"上傳:{m['name']}", path=d / "files", kind="upload",
                         created_at=m["created_at"])

    # ---------- 上傳 ----------
    def add_upload(self, files: list[tuple[str, bytes]]) -> tuple[Workspace, list[str]]:
        """files:[(檔名, 內容)]。只有一個 .zip 就解壓,否則當一般檔案。回傳 (資料夾, 被丟掉的檔名)。"""
        if not files:
            raise UploadRejected("沒有收到檔案")
        if sum(len(data) for _, data in files) > MAX_TOTAL_BYTES:
            raise UploadRejected("超過 20 MB 上限")
        if len(files) == 1 and files[0][0].lower().endswith(".zip"):
            entries, label = self._zip_entries(files[0][1]), files[0][0]
        else:
            entries = files
            label = files[0][0] if len(files) == 1 else f"{len(files)} 個檔案"
        if len(entries) > MAX_FILES:
            raise UploadRejected(f"檔案數超過 {MAX_FILES} 個")
        keep, skipped = [], []
        for name, data in entries:
            rel = _clean_rel(name)
            if rel is None:
                skipped.append(name)
                continue
            if len(data) > MAX_FILE_BYTES:
                raise UploadRejected(f"單檔超過 2 MB:{name}")
            keep.append((rel, data))
        if not keep:
            raise UploadRejected("過濾掉機密檔與系統檔之後,沒有可用的檔案")
        root = self.uploads_dir / f"up-{uuid.uuid4().hex[:8]}"
        for rel, data in keep:
            target = (root / "files").joinpath(*rel.parts)
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_bytes(data)
        (root / "meta.json").write_text(json.dumps({"name": label, "created_at": time.time()}, ensure_ascii=False),
                                        encoding="utf-8")
        return self._load_upload(root), skipped

    @staticmethod
    def _zip_entries(blob: bytes) -> list[tuple[str, bytes]]:
        try:
            zf = zipfile.ZipFile(io.BytesIO(blob))
        except zipfile.BadZipFile:
            raise UploadRejected("不是有效的 zip 檔") from None
        infos = [i for i in zf.infolist() if not i.is_dir()]
        if len(infos) > MAX_FILES:
            raise UploadRejected(f"檔案數超過 {MAX_FILES} 個")
        total, out = 0, []
        for info in infos:
            if (info.external_attr >> 16) & 0o170000 == 0o120000:
                raise UploadRejected(f"zip 裡有捷徑檔(符號連結),整包拒收:{info.filename}")
            total += info.file_size
            if total > MAX_TOTAL_BYTES:
                raise UploadRejected("解壓後超過 20 MB 上限")
            if info.file_size > MAX_FILE_BYTES:
                raise UploadRejected(f"單檔超過 2 MB:{info.filename}")
            out.append((info.filename, zf.read(info)))
        return out

    def cleanup(self, now: float | None = None) -> int:
        """刪掉過期或沒寫完的上傳;回傳刪了幾個。"""
        now = now if now is not None else time.time()
        if not self.uploads_dir.exists():
            return 0
        removed = 0
        for d in list(self.uploads_dir.iterdir()):
            w = self._load_upload(d)
            if w is None or now - w.created_at >= self.ttl:
                shutil.rmtree(d, ignore_errors=True)
                removed += 1
        return removed

    # ---------- 整份文件模式 ----------
    def whole_files(self, ws: Workspace) -> list[str]:
        """整份文件要附上的檔案:git 專案用 git ls-files,否則掃描;只留文字檔,排除機密檔與隱藏資料夾。"""
        root = ws.path
        if (root / ".git").exists():
            r = subprocess.run(["git", "-c", "core.quotepath=off", "ls-files"], cwd=root,
                               capture_output=True, text=True, encoding="utf-8")
            names = r.stdout.splitlines()
        else:
            names = [p.relative_to(root).as_posix() for p in root.rglob("*") if p.is_file()]
        out = []
        for name in sorted(names):
            p = PurePosixPath(name)
            if (p.suffix.lower() not in TEXT_SUFFIXES or _is_secret(Path(p.name))
                    or any(part.startswith(".") or part in {"node_modules", "__pycache__"} for part in p.parts[:-1])):
                continue
            out.append(name)
        return out
```

    id: str
    question: str
    mode: str
    options: dict
    status: str = "queued"
    events: list = field(default_factory=list)
    created_at: str = field(default_factory=_now)

    def summary(self) -> dict:
        return {"id": self.id, "question": self.question, "mode": self.mode, "options": self.options,
                "status": self.status, "created_at": self.created_at}


@dataclass
class Conversation:
    id: str
    title: str
    workspace: str
    created_at: str = field(default_factory=_now)
    turns: list = field(default_factory=list)
    messages: list = field(default_factory=list)  # messages_to_dict 的結果,追問時還原成前文

    def turn(self, tid: str) -> Turn:
        for t in self.turns:
            if t.id == tid:
                return t
        raise KeyError(tid)

    def summary(self) -> dict:
        status = self.turns[-1].status if self.turns else "done"
        return {"id": self.id, "title": self.title, "workspace": self.workspace,
                "created_at": self.created_at, "status": status}

    def detail(self) -> dict:
        return {**self.summary(), "turns": [t.summary() for t in self.turns]}

    @classmethod
    def from_dict(cls, data: dict) -> "Conversation":
        return cls(**{**data, "turns": [Turn(**t) for t in data.get("turns", [])]})


def _split_files(raw: str | None) -> list[str]:
    return [x.strip() for x in (raw or "").split(",") if x.strip()]


def _error_event(exc: Exception) -> dict:
    first = (str(exc).splitlines() or [""])[0][:300]
    if isinstance(exc, QuotaExceeded):
        return {"type": "error", "kind": "quota", "message": f"今天的額度不夠了:{first}"}
    if isinstance(exc, ContextTooLarge):
        return {"type": "error", "kind": "context", "message": f"內容太大:{first}"}
    if isinstance(exc, KeyError):
        return {"type": "error", "kind": "workspace",
                "message": "這個對話的資料夾已不存在(上傳的複本 24 小時後會刪除),請開新對話。"}
    if isinstance(exc, PermissionError):
        return {"type": "error", "kind": "permission", "message": first}
    if "[429]" in first or "[503]" in first:
        return {"type": "error", "kind": "busy", "message": "NVIDIA 服務忙線,已試過備援模型仍失敗,請稍後按重試。"}
    return {"type": "error", "kind": "other", "message": f"{type(exc).__name__}: {first}"}


class ConversationManager:
    def __init__(self, store_dir: Path, workspaces: WorkspaceRegistry, registry, quota,
                 executor: Executor | None = None, agent_fn: Callable[..., Any] = run_agent):
        self.store_dir = Path(store_dir)
        self.store_dir.mkdir(parents=True, exist_ok=True)
        self.workspaces = workspaces
        self.registry = registry
        self.quota = quota
        self.executor = executor or ThreadPoolExecutor(max_workers=2, thread_name_prefix="turn")
        self.agent_fn = agent_fn
        self.lock = threading.RLock()
        self.convs: dict[str, Conversation] = {}
        self.stop_flags: dict[str, threading.Event] = {}
        self._load()

    # ---------- 存取 ----------
    def _load(self) -> None:
        for path in sorted(self.store_dir.glob("*.json")):
            conv = Conversation.from_dict(json.loads(path.read_text(encoding="utf-8")))
            changed = False
            for turn in conv.turns:
                if turn.status not in FINISHED:  # 伺服器重開前沒跑完的,標成中斷
                    turn.status = "interrupted"
                    turn.events.append({"seq": len(turn.events), "t": _now(), "type": "end",
                                        "status": "interrupted", "steps": 0, "tokens": 0, "seconds": 0})
                    changed = True
            self.convs[conv.id] = conv
            if changed:
                self._save(conv)

    def _save(self, conv: Conversation) -> None:
        path = self.store_dir / f"{conv.id}.json"
        tmp = path.with_suffix(".tmp")
        tmp.write_text(json.dumps(asdict(conv), ensure_ascii=False), encoding="utf-8")
        os.replace(tmp, path)  # 先寫暫存檔再取代,寫到一半當機也不會壞檔

    def list(self) -> list[dict]:
        with self.lock:
            convs = list(self.convs.values())
        return [c.summary() for c in sorted(convs, key=lambda c: c.created_at, reverse=True)]

    def get(self, cid: str) -> Conversation:
        with self.lock:
            return self.convs[cid]

    # ---------- 建立與控制 ----------
    def start(self, workspace_id: str, question: str, mode: str, options: dict | None = None) -> tuple[str, str]:
        self.workspaces.get(workspace_id)  # 不存在就 KeyError
        self._validate(question, mode)
        conv = Conversation(id=uuid.uuid4().hex[:12], title=question.strip()[:30], workspace=workspace_id)
        with self.lock:
            self.convs[conv.id] = conv
        return conv.id, self.add_turn(conv.id, question, mode, options)

    def add_turn(self, cid: str, question: str, mode: str, options: dict | None = None) -> str:
        self._validate(question, mode)
        with self.lock:
            conv = self.convs[cid]
            if conv.turns and conv.turns[-1].status not in FINISHED:
                raise Busy("這個對話還有一輪在跑,等它結束或先按停止")
            turn = Turn(id=uuid.uuid4().hex[:8], question=question.strip(), mode=mode, options=dict(options or {}))
            conv.turns.append(turn)
            self.stop_flags[turn.id] = threading.Event()
            self._save(conv)
        self.executor.submit(self._run, cid, turn.id)
        return turn.id

    def cancel(self, cid: str, tid: str) -> None:
        self.get(cid).turn(tid)  # 不存在就 KeyError
        flag = self.stop_flags.get(tid)
        if flag:
            flag.set()

    @staticmethod
    def _validate(question: str, mode: str) -> None:
        if not (question or "").strip():
            raise ValueError("問題不能是空的")
        if mode not in MODES:
            raise ValueError(f"模式只能是 {sorted(MODES)}")

    # ---------- 執行 ----------
    def _emit(self, conv: Conversation, turn: Turn, event: dict) -> None:
        with self.lock:
            turn.events.append({"seq": len(turn.events), "t": _now(), **event})
            self._save(conv)

    def _finish(self, conv: Conversation, turn: Turn, status: str, steps: int, tokens: int, started: float) -> None:
        with self.lock:
            turn.status = status
            self._emit(conv, turn, {"type": "end", "status": status, "steps": steps, "tokens": tokens,
                                    "seconds": round(time.monotonic() - started, 1)})
        self.stop_flags.pop(turn.id, None)

    def _run(self, cid: str, tid: str) -> None:
        conv = self.get(cid)
        turn = conv.turn(tid)
        stop = self.stop_flags[tid]
        started = time.monotonic()
        if stop.is_set():  # 還在排隊就被取消
            self._finish(conv, turn, "cancelled", 0, 0, started)
            return
        with self.lock:
            turn.status = "running"
            self._save(conv)
        try:
            ws = self.workspaces.get(conv.workspace)
            whole = turn.mode == "whole"
            history = messages_from_dict(conv.messages) if conv.messages else None
            result = self.agent_fn(
                turn.question, registry=self.registry, quota=self.quota, user="web",
                attachments=self.workspaces.whole_files(ws) if whole else _split_files(turn.options.get("files")),
                role=turn.options.get("role") or None, thinking=turn.options.get("thinking"),
                whole=whole, workspace=ws.path, history=history,
                on_event=lambda e: self._emit(conv, turn, e), should_stop=stop.is_set)
            with self.lock:
                conv.messages = messages_to_dict(result.messages)
            self._finish(conv, turn, "cancelled" if result.stopped else "done",
                         result.steps, result.tokens_used, started)
        except Exception as exc:  # 任何錯誤都要讓畫面知道,不能讓這一輪永遠停在「執行中」
            self._emit(conv, turn, _error_event(exc))
            self._finish(conv, turn, "failed", 0, 0, started)
```

"""FastAPI 路由。只聽本機(見 __main__.py),沒有登入。"""
from __future__ import annotations

import asyncio
import json
from pathlib import Path

from fastapi import FastAPI, File, HTTPException, Request, UploadFile
from fastapi.responses import FileResponse, Response, StreamingResponse
from fastapi.staticfiles import StaticFiles
from langchain_core.messages import HumanMessage, SystemMessage
from pydantic import BaseModel

from agent import SYSTEM_PROMPT, build_user_prompt
from token_guard import ContextTooLarge, message_tokens
from web.conversations import Busy, ConversationManager
from web.workspaces import UploadRejected, WorkspaceRegistry

STATIC = Path(__file__).with_name("static")


class TurnIn(BaseModel):
    question: str
    mode: str = "fix"
    role: str | None = None
    thinking: bool | None = None
    files: str | None = None

    def options(self) -> dict:
        return {"role": self.role, "thinking": self.thinking, "files": self.files}


class ConversationIn(TurnIn):
    workspace: str


def create_app(manager: ConversationManager, workspaces: WorkspaceRegistry, registry,
               poll_seconds: float = 0.3) -> FastAPI:
    app = FastAPI(title="研發助理")
    app.state.manager = manager
    app.mount("/static", StaticFiles(directory=STATIC, check_dir=False), name="static")

    def conversation_or_404(cid: str):
        try:
            return manager.get(cid)
        except KeyError:
            raise HTTPException(404, "沒有這個對話") from None

    def workspace_or_404(wid: str):
        try:
            return workspaces.get(wid)
        except KeyError:
            raise HTTPException(404, "沒有這個資料夾") from None

    @app.get("/")
    def index() -> FileResponse:
        return FileResponse(STATIC / "index.html")

    @app.get("/api/workspaces")
    def list_workspaces() -> list[dict]:
        return [w.to_dict() for w in workspaces.list()]

    @app.post("/api/uploads")
    async def upload(files: list[UploadFile] = File(...)) -> dict:
        blobs = [(f.filename or "file", await f.read()) for f in files]
        try:
            ws, skipped = workspaces.add_upload(blobs)
        except UploadRejected as e:
            raise HTTPException(400, str(e)) from None
        return {"workspace": ws.to_dict(), "skipped": skipped}

    @app.get("/api/estimate")
    def estimate(workspace: str) -> dict:
        ws = workspace_or_404(workspace)
        files = workspaces.whole_files(ws)
        prompt = build_user_prompt("", files, registry.limits["inline_file_max_tokens"], True, root=ws.path)
        tokens = message_tokens([SystemMessage(SYSTEM_PROMPT), HumanMessage(prompt)])
        limit = registry.budget(registry.routing["long_context_role"])
        try:
            role = registry.pick_role(prompt_tokens=tokens, needs_tools=False)
        except ContextTooLarge:
            return {"files": len(files), "tokens": tokens, "limit": limit, "ok": False, "role": None, "model": None}
        return {"files": len(files), "tokens": tokens, "limit": limit, "ok": True,
                "role": role, "model": registry.spec(role).model}

    @app.get("/api/conversations")
    def list_conversations() -> list[dict]:
        return manager.list()

    @app.post("/api/conversations")
    def new_conversation(body: ConversationIn) -> dict:
        try:
            cid, tid = manager.start(body.workspace, body.question, body.mode, body.options())
        except KeyError:
            raise HTTPException(400, "沒有這個資料夾") from None
        except ValueError as e:
            raise HTTPException(400, str(e)) from None
        return {"conversation_id": cid, "turn_id": tid}

    @app.get("/api/conversations/{cid}")
    def get_conversation(cid: str) -> dict:
        return conversation_or_404(cid).detail()

    @app.post("/api/conversations/{cid}/turns")
    def add_turn(cid: str, body: TurnIn) -> dict:
        conversation_or_404(cid)
        try:
            tid = manager.add_turn(cid, body.question, body.mode, body.options())
        except Busy as e:
            raise HTTPException(409, str(e)) from None
        except ValueError as e:
            raise HTTPException(400, str(e)) from None
        return {"turn_id": tid}

    @app.post("/api/conversations/{cid}/turns/{tid}/cancel")
    def cancel(cid: str, tid: str) -> dict:
        try:
            manager.cancel(cid, tid)
        except KeyError:
            raise HTTPException(404, "沒有這一輪") from None
        return {"ok": True}

    @app.get("/api/conversations/{cid}/turns/{tid}/events")
    async def events(cid: str, tid: str, request: Request, after: int = -1) -> Response:
        conv = conversation_or_404(cid)
        try:
            turn = conv.turn(tid)
        except KeyError:
            raise HTTPException(404, "沒有這一輪") from None
        last = request.headers.get("last-event-id", "")
        start = int(last) + 1 if last.isdigit() else after + 1
        if turn.events and turn.events[-1]["type"] == "end" and start > turn.events[-1]["seq"]:
            return Response(status_code=204)  # 已經全部送完;回 204 瀏覽器的 EventSource 才不會一直自動重連

        async def stream():
            i = start
            while True:
                evs = turn.events
                while i < len(evs):
                    event = evs[i]
                    i += 1
                    yield f"id: {event['seq']}\nevent: {event['type']}\ndata: {json.dumps(event, ensure_ascii=False)}\n\n"
                    if event["type"] == "end":
                        return
                if evs and evs[-1]["type"] == "end":  # 斷線重連時結束事件已經送過,不要空等
                    return
                if await request.is_disconnected():
                    return
                await asyncio.sleep(poll_seconds)

        return StreamingResponse(stream(), media_type="text/event-stream",
                                 headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"})

    return app

function el(tag, attrs = {}, ...children) {
  const node = document.createElement(tag);

    if (value == null) continue;
    if (key === "class") node.className = value;
    else if (key === "text") node.textContent = value;
    else if (key.startsWith("on")) node.addEventListener(key.slice(2), value);
    else node.setAttribute(key, value);

async function api(path, options = {}) {

  if (!res.ok) throw new Error(typeof data.detail === "string" ? data.detail : `伺服器回應 ${res.status}`);
  return data;
}

// ---------- 事件 ----------

function streamTurn(conv, turn, view, live) {
  const source = new EventSource(`/api/conversations/${conv.id}/turns/${turn.id}/events`);
  state.streams.set(turn.id, source);
  // 連線失敗時瀏覽器也會發一個沒有 data 的 "error" 事件,和我們自己的 error 事件同名,要先過濾
  const on = (type, handler) => source.addEventListener(type, (e) => {
    if (typeof e.data === "string") handler(JSON.parse(e.data));
  });
  on("start", (d) => {
    view.model = modelName(d.model);
    view.startedAt = Date.parse(d.t);
    setMeta(view, `${view.model} · 預估 ${fmt(d.est_tokens)} tokens`);
    startTimer(view);
  });
  on("step", (d) => {
    view.model = modelName(d.model);
    setPending(view, `第 ${d.step} 步:等 ${view.model} 回應…`);
  });
  on("tool_call", (d) => addToolRow(view, d));
  on("tool_result", (d) => fillToolRow(view, d));
  on("model_switch", (d) => {
    clearPending(view);
    view.steps.append(el("li", { class: "notice", text: `${d.reason},改用 ${d.to_role}` }));
  });
  on("patch", (d) => addPatchCard(conv, view, d));
  on("answer", (d) => showAnswer(view, d.text, live));
  on("error", (d) => showError(view, d));
  on("end", (d) => {
    source.close();
    state.streams.delete(turn.id);
    finishTurn(view, d);
  });
  source.onerror = () => {
    if (source.readyState === EventSource.CLOSED) {
      stopTimer(view);
      view.meta.textContent = "連線中斷,重新整理頁面可以接續";
    }
  };
}

function setMeta(view, text) {
  view.metaText = text;
  renderMeta(view);
}

function renderMeta(view) {
  const secs = view.startedAt ? Math.max(0, Math.round((Date.now() - view.startedAt) / 1000)) : null;
  view.meta.textContent = secs == null ? view.metaText : `${view.metaText} · ${secs} 秒`;
}

function startTimer(view) {
  stopTimer(view);
  view.timer = setInterval(() => renderMeta(view), 1000);
}

function stopTimer(view) {
  if (view.timer) clearInterval(view.timer);
  view.timer = null;
}

function setPending(view, text) {
  clearPending(view);
  view.pending = el("li", { class: "pending", text });
  view.steps.append(view.pending);
}

function clearPending(view) {
  view.pending?.remove();
  view.pending = null;
}

  }
  if (name === "search_code") return `"${a.pattern}"${a.glob ? ` · ${a.glob}` : ""}`;
  if (name === "run_tests") return a.target || "tests";
  if (name === "propose_patch") return a.path;
  return JSON.stringify(a).slice(0, 80);
}

function testChip(text, prefix = "") {
  const failed = /(\d+) failed/.exec(text);
  const passed = /(\d+) passed/.exec(text);
  if (failed) return ["bad", `${prefix}${failed[1]} failed`];
  if (passed) return ["good", `${prefix}${passed[1]} passed`];
  return null;
}

function addToolRow(view, d) {
  clearPending(view);
  const chip = el("span", { class: "chip muted", text: "執行中" });
  const detail = el("pre", { class: "tool-output", hidden: "" });
  const head = el("button", { type: "button", class: "tool-head", onclick: () => { detail.hidden = !detail.hidden; } },
    el("span", { class: "tool-name", text: TOOL_LABEL[d.name] || d.name }),
    el("code", { text: describeArgs(d.name, d.args) }),
    chip);
  view.steps.append(el("li", { class: "tool" }, head, detail));
  view.rows.push({ step: d.step, name: d.name, chip, detail, done: false });
}

function fillToolRow(view, d) {
  const row = view.rows.find((r) => !r.done && r.step === d.step && r.name === d.name);
  if (!row) return;
  row.done = true;
  row.detail.textContent = d.preview;
  const isTest = d.name === "run_tests" || d.name === "propose_patch";
  const test = isTest ? testChip(d.preview, d.name === "propose_patch" ? "套用後 " : "") : null;
  const [cls, text] = !d.ok ? ["bad", "錯誤"] : test || ["muted", "完成"];
  row.chip.className = `chip ${cls}`;
  row.chip.textContent = text;
}

function addPatchCard(conv, view, d) {
  const diff = el("pre", { class: "diff" });
  for (const line of d.diff.split("\n")) {
    const cls = line.startsWith("+") && !line.startsWith("+++") ? "add"
      : line.startsWith("-") && !line.startsWith("---") ? "del" : null;
    diff.append(el("span", { class: cls }, `${line}\n`));
  }
  const [cls, text] = testChip(d.test_output, "套用後 ")
    || (d.passed ? ["good", "套用後測試通過"] : ["bad", "套用後測試失敗"]);
  const actions = el("div", { class: "patch-actions" }, el("span", { class: `chip ${cls}`, text }));
  const ws = workspaceById(conv.workspace);
  if (ws && ws.kind === "config" && d.patch_file) {
    const cmd = `git -C "${ws.path}" apply "${d.patch_file}"`;
    actions.append(el("button", { type: "button", text: "複製 git apply 指令",
      onclick: (e) => copyText(cmd, e.currentTarget) }));
  } else {
    actions.append(el("span", { class: "note", text: "這是上傳的複本,patch 不能直接套回你的專案" }));
  }
  view.extras.append(el("div", { class: "patch" }, el("code", { text: d.path }), diff, actions));
}

async function copyText(text, button) {
  try {
    await navigator.clipboard.writeText(text);
    button.textContent = "已複製";
  } catch {
    window.prompt("複製這行指令:", text);
  }
}

async function sendTurn(body) {
  if (state.current) {
    const r = await api(`/api/conversations/${state.current.id}/turns`, { method: "POST", body: JSON.stringify(body) });
    addTurn(state.current, { id: r.turn_id, question: body.question, mode: body.mode, options: body, status: "queued" }, true);
  } else {
    const r = await api("/api/conversations", {
      method: "POST", body: JSON.stringify({ ...body, workspace: $("#workspace").value }) });
    await openConversation(r.conversation_id, r.turn_id);
  }
}

async function onSubmit(event) {
  event.preventDefault();
  if (state.running) {
    const { cid, tid } = state.running;
    try {
      await api(`/api/conversations/${cid}/turns/${tid}/cancel`, { method: "POST" });
      hint("已要求停止:正在等模型回應的那一步回來後就會停。");
    } catch (e) { hint(e.message, true); }
    return;
  }
  const question = $("#question").value.trim();
  if (!question) { hint("先輸入問題", true); return; }
  if (!state.current && !$("#workspace").value) { hint("先選一個資料夾", true); return; }
  const body = { question, mode: currentMode(), role: $("#role").value || null,
                 thinking: thinkingValue(), files: $("#files").value.trim() || null };
  try {
    await sendTurn(body);
    $("#question").value = "";
    hint("");
  } catch (e) { hint(e.message, true); }
}

async function onUpload(event) {
  const files = [...event.target.files];
  event.target.value = "";
  if (!files.length) return;
  const form = new FormData();
  for (const f of files) form.append("files", f, f.name);
  hint("上傳中…");
  try {
    const r = await api("/api/uploads", { method: "POST", body: form });
    if (state.current) newChat();
    await loadWorkspaces(r.workspace.id);
    const skipped = r.skipped.length ? `;略過 ${r.skipped.length} 個機密或系統檔:${r.skipped.slice(0, 5).join("、")}` : "";
    hint(`已上傳「${r.workspace.name}」${skipped}。這是複本,24 小時後刪除,patch 不能直接套回你的專案。`);
    updateEstimate();
  } catch (e) { hint(e.message, true); }
}

async function updateEstimate() {
  const send = $("#send");
  if (!state.running) send.disabled = false;
  if (currentMode() !== "whole") return;
  const ws = state.current ? state.current.workspace : $("#workspace").value;
  if (!ws) return;
  hint("估算整份文件大小中…");
  try {
    const e = await api(`/api/estimate?workspace=${encodeURIComponent(ws)}`);
    if (e.ok) {
      const wait = e.tokens > 100000 ? "可能要好幾分鐘,可以先去做別的事" : "大約一兩分鐘";
      hint(`整份文件:${fmt(e.files)} 個檔案,約 ${fmt(e.tokens)} tokens → 會用 ${modelName(e.model)},${wait}。`);
    } else {
      hint(`約 ${fmt(e.tokens)} tokens,超過上限 ${fmt(e.limit)},請改用較小的資料夾。`, true);
      if (!state.running) send.disabled = true;
    }
  } catch (err) { hint(err.message, true); }
}

async function init() {
  $("#new-chat").addEventListener("click", newChat);

  $("#question").addEventListener("keydown", (e) => {
    // isComposing:注音 / 倉頡選字時按 Enter 是確認選字,不是送出
